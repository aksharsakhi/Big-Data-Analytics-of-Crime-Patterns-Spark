Scala RDD Core Implementation Snippet

Extract from src/main/scala/bigdata/spark_rdd_script.scala 
val rawRdd = sc.textFile("dataset/chicago_crimes_clean.csv")
val header = rawRdd.first()

// Transformation 1: filter (narrow - discard header and malformed records)
val dataRdd = rawRdd.filter(_ != header)

// Transformation 2: map (deserialize CSV into typed CrimeRecord)
val crimesRdd = dataRdd.map(parseCsvLine).persist(StorageLevel.MEMORY_AND_DISK)

// Key-Value Transformation: reduceByKey for hotspot district ranking
val districtCounts = crimesRdd.map(c => (c.district, 1)).reduceByKey(_ + _).sortBy(_._2, ascending = false)

// Actions: count(), take(5)
val total = crimesRdd.count()
val topHotspots = districtCounts.take(5)
